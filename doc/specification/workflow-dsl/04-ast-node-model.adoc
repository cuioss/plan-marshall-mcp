package de.cuioss.pm.mcp.core.workflow.ast;

import java.util.List;
import java.util.Map;
import java.util.Optional;
import java.util.Set;

import de.cuioss.pm.mcp.core.workflow.spi.OutcomeClass;

public record SourceLocation(String resourceName, int line, int column) {}

public enum WorkflowScope { PLAN, EPIC, WORKSPACE, ENTITY, SUBWORKFLOW }

/** Closed link relations (hypermedia-format.adoc, Closed Link Relations): allowed forms and the default form. */
public enum LinkRelation {
    NEXT(Set.of(LinkForm.NOOP, LinkForm.SUBMIT), LinkForm.NOOP),
    ALT(Set.of(LinkForm.NOOP, LinkForm.SUBMIT), LinkForm.NOOP),
    RETRY(Set.of(LinkForm.NOOP), LinkForm.NOOP),
    FIX(Set.of(LinkForm.NOOP, LinkForm.SUBMIT, LinkForm.DISPATCH), LinkForm.NOOP),
    LOOP_BACK(Set.of(LinkForm.NOOP, LinkForm.SUBMIT), LinkForm.SUBMIT),
    SUBMIT(Set.of(LinkForm.SUBMIT), LinkForm.SUBMIT),
    DISPATCH(Set.of(LinkForm.DISPATCH), LinkForm.DISPATCH),
    WAIT(Set.of(LinkForm.WAIT), LinkForm.WAIT),
    ESCALATE(Set.of(LinkForm.ESCALATE), LinkForm.ESCALATE),
    ABORT(Set.of(LinkForm.NOOP, LinkForm.SUBMIT), LinkForm.NOOP),
    EMIT(Set.of(LinkForm.NOOP, LinkForm.SUBMIT), LinkForm.NOOP);

    private final Set<LinkForm> allowedForms;
    private final LinkForm defaultForm;
    LinkRelation(Set<LinkForm> allowedForms, LinkForm defaultForm) {
        this.allowedForms = allowedForms;
        this.defaultForm = defaultForm;
    }
    public Set<LinkForm> allowedForms() { return allowedForms; }
    /** Form of a `link` clause that omits `[form]`. */
    public LinkForm defaultForm() { return defaultForm; }
}

/** Closed link forms (hypermedia-format.adoc, The Five Normative Link Forms). */
public enum LinkForm { SUBMIT, WAIT, DISPATCH, ESCALATE, NOOP }

public record StateMigration(String oldStateId, String newStateId, SourceLocation location) {}

public sealed interface ParamType

    record Text(Optional<Integer> minLength, Optional<Integer> maxLength) implements ParamType {}
    record Int(Optional<Integer> min, Optional<Integer> max) implements ParamType {}
    record EnumOf(List<String> values) implements ParamType {}
    /** `schema[<schemaName>]`: a JSON object validated against the named closed schema (hypermedia-format.adoc, Named Parameter Schemas). */
    record SchemaRef(String schemaName) implements ParamType {}
    record ListOf(ParamType elementType) implements ParamType {}   // elementType is never a ListOf
}

/** Free parameter of a link (`param_decl`). */
public record ParamSpec(String name, ParamType type, boolean optional, SourceLocation location) {}

/** Sub-workflow parameter declared in the header (`param <name>: <type>`); always required. */
public record ParamDecl(String name, ParamType type, SourceLocation location) {}

    WorkflowScope scope,
    List<ParamDecl> params,          // non-empty only when scope == SUBWORKFLOW

    int formatVersion,                 // DSL format of every file of the unit (all files declare the same)
    UnitOrigin origin,                 // DEFAULT (compiled) or PROJECT (.marshall/config/)

    List<StateMigration> migrations,

public record TerminalOutcome(String id, OutcomeClass outcomeClass) {}

public record StateNode(

    Optional<String> skillPath,
    Optional<String> factsPrimitiveId,
    StateBehavior behavior,
    SourceLocation location
) {}

public sealed interface StateBehavior
    permits StateBehavior.InteractiveLinks,

    record JobStep(String primitiveId, boolean composite, List<OutcomeTransition> transitions) implements StateBehavior {}
    record SubWorkflowCall(

        List<OutcomeTransition> transitions

    record TerminalState() implements StateBehavior {}
}

public record CallFrame(

    String callerStateId,
    String callerPhase,
    String subWorkflowId,
    Map<String, String> boundArgs,
    String pushedAt

public record LinkSpec(

    String idPattern,
    LinkForm form,
    List<ParamSpec> freeParams,
    Optional<GuardSpec> guard,
    LinkAction action,
    List<OutcomeTransition> transitions,

public sealed interface LinkAction
    permits LinkAction.ExecutePrimitive,
            LinkAction.DispatchSubAgent,
            LinkAction.EscalateDecision {

    record ExecutePrimitive(String primitiveId) implements LinkAction {}
    /** `dispatch <entity_kind> role <role_key>`; the effort level is always resolved from the role. */
    record DispatchSubAgent(String targetEntityKind, String roleKey) implements LinkAction {}
    record EscalateDecision(String decisionId) implements LinkAction {}
}

public sealed interface GuardExpr
    permits GuardExpr.InWorktree,
            GuardExpr.HasOpenFindings,
            GuardExpr.AllTasksCompleted,
            GuardExpr.CustomPredicate,
            GuardExpr.And,
            GuardExpr.Or,
            GuardExpr.Not {

    record InWorktree() implements GuardExpr {}
    record HasOpenFindings() implements GuardExpr {}
    record AllTasksCompleted() implements GuardExpr {}
    record CustomPredicate(String primitiveId) implements GuardExpr {}
    record And(GuardExpr left, GuardExpr right) implements GuardExpr {}
    record Or(GuardExpr left, GuardExpr right) implements GuardExpr {}
    record Not(GuardExpr operand) implements GuardExpr {}
}

/** Result of evaluating a guard; a transition fires only on HELD. */
public sealed interface GuardResult
    permits GuardResult.Held, GuardResult.NotHeld, GuardResult.CouldNotEvaluate {

    record Held() implements GuardResult {}
    record NotHeld() implements GuardResult {}
    record CouldNotEvaluate(String reason) implements GuardResult {}
}

public record GuardSpec(GuardExpr expr, List<String> dependsOnConfig) {}

public record StepCompletion(String stepId, StepCompletionStatus status, SourceLocation location) {}

[#guard-results]
==== Three-Valued Guard Results
_Implementation watch: link:../../implementation-watch/workflow-dsl.adoc#PM-WATCH-DSL-2[PM-WATCH-DSL-2]_

Every guard evaluates to the sealed `GuardResult`: `HELD`, `NOT_HELD`, or `COULD_NOT_EVALUATE(reason)` (a built-in guard whose input cannot be read, a `predicate:` primitive that fails or times out, an unreadable configuration key). A link is offered, and its transition fires, only on `HELD`; `NOT_HELD` omits the link (`PM-WF-3`). `COULD_NOT_EVALUATE` is never folded into either: the engine raises the `guard_indeterminate` condition on the current state (<<condition-overlays>>), naming the guard, the link, and the reason in `facts`, and the undecided link is neither offered nor refused silently. `And`, `Or`, and `Not` propagate `COULD_NOT_EVALUATE` unless the other operand decides the result (`NOT_HELD` and anything is `NOT_HELD`, `HELD` or anything is `HELD`).


[#condition-overlays]
=== Condition Overlays (Engine-Owned)

A condition is an engine-owned overlay on the scope's current state, not a state: `blocked` (an operator decision is pending), `footprint_violation`, `dispatch_incomplete`, `config_drift`, `store_tampered`, and `guard_indeterminate` (link:../state-catalogue/01-conventions-generic-conditions.adoc#state-catalogue-conditions[State Catalogue, Conditions]). The engine raises it where the condition arises (an `escalate` link without elicitation, the footprint check of `git.stage-and-commit` (xref:06-reusable-sub-workflows.adoc#_six_stage_execution_contract_of_git_commit_and_push[Six-Stage Execution Contract of `git.commit_and_push`], Stage 4) or of a task `submit` (xref:05-primitives.adoc#independent-transition-validation[Independent Server-Side Transition & Parameter Validation (`PM-WF-6`)] item 6), a planning transition guard, a dispatch return, a phase handshake, a locked digest check, a guard that could not be evaluated (<<guard-results>>)) and records it in `mcp/status.json` as `pending_condition` (`kind` from the closed `ConditionKind` enumeration, `decision_id` when an operator decision is pending, `facts`, `raised_at`, link:../store-schemas.adoc[Store Schemas Specification]). `state`, `phase`, and `call_stack` are not changed, so a condition raised inside a sub-workflow leaves the scope inside that sub-workflow.

* *Rendering*: While a condition is set, the representation keeps `state` as the current state id, carries the field `condition` naming the kind (link:../hypermedia-format/01-representation-and-link-forms.adoc#canonical-top-level-fields[Hypermedia Format Specification]) and the condition's `facts`, and offers only the condition's links (`wait:decision:<decision_id>` for a pending decision; `fix` and `escalate` for a footprint violation; `retry` and `escalate` for an indeterminate guard; the links the condition names otherwise), plus the implicit unrendered `report_bypass`. Every other declared or implicit link of the state is answered with `link_not_available`.
* *Resolution Routing*: A decision resolution is routed like a primitive outcome of the state that raised the condition: for an `escalate` link, the resolution is the `DecisionSpec` outcome of that link and takes its declared outcome transition in the current state's unit (Rule 2 guarantees the bijection), in the same transaction that clears the condition; `call_stack` is untouched, so a resolution inside a sub-workflow stays inside it and only a sub-workflow terminal returns to the caller. A condition raised by the engine rather than a link (a guard, a credential check, a footprint check, a dispatch return, a phase handshake) is cleared by its own resolution (the operator's decision, a `fix` whose next footprint check passes, `resume:<entity>`, `adopt` / `keep`, a `retry` whose re-evaluation of every guard of the state yields `HELD` or `NOT_HELD`) and re-renders the current state with its declared links.
* *Not Declared in DSL Units*: Conditions are never declared in a `*.pmwf` unit and are never a transition target. A DSL transition whose target names a condition or a former engine-implicit state id (`<phase>.blocked`, `<phase>.footprint-violation`, `<phase>.dispatch-incomplete`, `<phase>.config-drift`) fails validation (Rule 3).

[#static-verification-rules]
=== Static AST Verification Rules (Build Time & Every Unit Load)
_See Requirement link:../../Requirements.adoc#PM-WF-5[PM-WF-5: Sub-Agent Dispatch & Single-Level Leaf Constraint]_

_See Requirement link:../../Requirements.adoc#PM-WF-16[PM-WF-16: Reusable Compound Sub-Workflows & Config-Driven Git Commit/Push]_

_See Requirement link:../../Requirements.adoc#PM-MIG-4[PM-MIG-4: Stage Exit Criteria & Lifecycle Gate]_

_Implementation watch: link:../../implementation-watch/workflow-dsl.adoc#PM-WATCH-DSL-1[PM-WATCH-DSL-1], link:../../implementation-watch/workflow-dsl.adoc#PM-WATCH-DSL-3[PM-WATCH-DSL-3], link:../../implementation-watch/workflow-dsl.adoc#PM-WATCH-DSL-4[PM-WATCH-DSL-4], link:../../implementation-watch/workflow-dsl.adoc#PM-WATCH-DSL-5[PM-WATCH-DSL-5], link:../../implementation-watch/workflow-dsl.adoc#PM-WATCH-DSL-8[PM-WATCH-DSL-8], link:../../implementation-watch/workflow-dsl.adoc#PM-WATCH-DSL-9[PM-WATCH-DSL-9], link:../../implementation-watch/workflow-dsl.adoc#PM-WATCH-DSL-11[PM-WATCH-DSL-11]_

`WorkflowValidator` verifies every `WorkflowDefinition` in JUnit build tests (the default units) and on every load of a project's units (<<project-owned-units>>), failing closed if any rule is violated; a project unit is additionally checked against the compiled invariants. A `WorkflowDefinition` is always a complete definition unit (xref:02-units-jobs-decisions.adoc#workflow-definition-units[Workflow Definition Units (Multi-File Workflows)]): the merged union of all phase files of the plan workflow, a single-file top-level workflow, or one sub-workflow. Rules 2, 3, 5, 8, 9, 10, and 11 run over that union, never over an individual phase file:

1. *Primitive Binding & Parameter Equivalence*: Every `primitiveId` in `do`, `auto`, `job`, `facts`, and `predicate:` must resolve to a registered `Primitive<P, R>`. For every `LinkSpec`, the declared `freeParams` list must exactly match the name, type, and constraints of the `Record` components annotated as `@FreeParam` on `P`. Every `ParamType.SchemaRef` (`schema[<schema_name>]`, in a link's free parameters or a `DecisionSpec.escalateParams`) must name a schema registered in the Named Parameter Schemas (link:../hypermedia-format/02-link-grammar.adoc#named-parameter-schemas[Hypermedia Format Specification]); an unknown schema name fails validation with the source file and line. Every primitive that declares `enqueuesJob()` must declare the outcome `started`, and every `started` transition of such a primitive must target a `job` or `composite_job` state, which binds the returned job id instead of starting a job (xref:02-units-jobs-decisions.adoc#job-state-binding[Job State Entry & Binding (`JobStep`)]); no other outcome of such a primitive may target a job state.
2. *Exhaustive Outcome Coverage*: For every `auto`, `job`, `composite_job`, and `link` action backed by `Primitive<P, R>`, the set of `OutcomeTransition.outcomeName` entries must form an exact 1-to-1 bijection with the enum constants of `R` (`R.getEnumConstants()`). For `job` and `composite_job` states this includes `cancelled`, the outcome of the implicit `cancel:<job_id>` link (xref:02-units-jobs-decisions.adoc#implicit-job-links[Implicit Links of Job States]). For every `link` whose action is `escalate <decision>`, the decision kind must be registered in the Decision Registry (xref:02-units-jobs-decisions.adoc#decision-registry[Decision Registry & Typed Escalation Outcomes]), and the `OutcomeTransition.outcomeName` entries must form the same exact bijection with the `DecisionSpec.outcomes` of that kind: every declared outcome routed exactly once, no undeclared outcome. An unregistered decision kind, a missing outcome, and an unknown outcome name fail validation with the source file and line number.
3. *Graph Integrity*: State identifiers must be unique across all files of the definition unit. `startState` and every `OutcomeTransition.targetStateId` must exist in the unit's `states` (a cross-file target within the plan workflow is valid; a target inside another unit or inside a sub-workflow is not). Every `StateNode` must be reachable from the unit's `startState`, and reachability is checked for every combination of the values of the configuration keys that the unit's guards declare in `depends_on_config` (every enum value, `true` and `false` for a boolean): a state or outcome transition that no combination reaches fails validation, and one that some combinations do not reach is reported with those combinations. Every non-terminal `StateNode` must have at least one outgoing transition path to a `terminal` state; in a `scope workspace` unit, which declares no terminal state because the workspace scope lives as long as the project's adoption, every state must instead have a path back to the unit's `startState`. Every `skill` clause must resolve to a vocabulary skill in the compiled corpus, and every link identifier pattern must be unique within its state, including the implicit `wait:<job_id>` and `cancel:<job_id>` links of job states. Condition overlays (<<condition-overlays>>) are engine-owned and need no target in `states`: a transition whose target names a condition kind or a former engine-implicit state id (`<phase>.blocked`, `<phase>.footprint-violation`, `<phase>.dispatch-incomplete`, `<phase>.config-drift`) fails validation.
4. *Single-Level Leaf Sub-Agent Constraint (`PM-WF-5`)*: Any `WorkflowDefinition` with `scope == WorkflowScope.ENTITY` (or any state reachable within an entity sub-agent scope) is strictly forbidden from containing a `LinkSpec` with `form == LinkForm.DISPATCH` or `LinkAction.DispatchSubAgent`. Sub-agents are leaves and can never dispatch nested sub-agents.
5. *Backwards-Compatible State Migration (`PM-MIG-4`)*: Every state name present in the committed `released-states.txt` baseline must either exist in `states` or appear as `oldStateId` in a `migrate <old> -> <new>` declaration whose target exists in `states`. The rule applies to the default units; an in-flight plan of a project whose unit dropped or renamed a state stays on its bound unit copy (<<project-owned-units>>, item 4).
6. *Sub-Workflow Outcome Bijection & Parameter Binding (`PM-WF-16`)*: Every `StateBehavior.SubWorkflowCall` (`call <subWorkflowId>`) must reference a registered `WorkflowDefinition` whose `scope == WorkflowScope.SUBWORKFLOW`. Every terminal `StateNode` in a `SUBWORKFLOW` declares its `terminalOutcome` (`state <id> terminal <outcome_id> <outcome_class>`, which the grammar requires of every terminal state), and the set of `OutcomeTransition.outcomeName` entries on each `call` site must form an exact 1-to-1 bijection with the target sub-workflow's declared `terminalOutcome` values. The `paramBindings` of every `call` site are checked against the target's `params` (`param` declarations): every declared parameter must be bound exactly once, no binding may name an undeclared parameter, and every bound value must match the declared type (membership in the `enum[…]` value set, or an integer within the declared `int` range). `phase inherit` is admitted only in a `SUBWORKFLOW` unit. A `call` site's transition for a terminal outcome of class `aborted` must target a `link` state of the caller's unit (a decision state, or a state offering the escalation that ends the scope), never a terminal state and never a state that continues the flow.
7. *Acyclic Call-Graph DAG & Bounded Depth (`PM-WF-16`)*: Across all registered `WorkflowDefinition` instances, the static `call` graph must form a strict directed acyclic graph (DAG) with zero cycles and a maximum static call-stack depth $\le 4$.
8. *Gate Artifact Producer Coverage*: For every guard or gating primitive that declares a required artifact (`requiredArtifacts()`, for example `pr_title`, `outline.md`, or `execution-manifest.json`), every path from the unit's `startState` to that guard or state, including recipe and lane shortcuts, must pass a state whose primitive declares the artifact in `producedArtifacts()`. A path without a producer fails validation, so a gate can never refuse for an artifact its lane could not produce.
9. *Step Completion Exactly Once & Failure Outcomes Never Map to Success*: Every outcome constant of every primitive and `composite_job` outcome set declares an `OutcomeClass` (xref:05-primitives.adoc#primitive-outcome-classes[Outcome Classes & Step Records]), and every terminal state declares its class in `terminal <outcome_id> <outcome_class>`; a sub-workflow's terminal class is the class of the matching outcome at its `call` sites. A step is completed only by a `completes <step_id> as done|skipped|failed|skipped_by_operator` clause (xref:01-module-boundary-and-grammar.adoc#_formal_ebnf_grammar_for_the_workflow_dsl[Formal EBNF Grammar for the Workflow DSL]). The rule checks:
** *Exactly once*: Every finalize step named by a `completes` clause is completed exactly once on every path from the state that begins it (the state `finalize.next-step` selects for it, or the region entry state for steps that share one, such as `finalize.review-cap-check` for the `22..40` wait region) to the state of order `70` (`finalize.merge-gate`) or to a terminal state. A path ends early at an `UNSETTLED` outcome (`loop_back`, defer) or an `ABORTED` outcome (a sub-workflow returning `aborted`), which hands the step back without a completion. Entering `plan.aborted` or `epic.aborted` records every begun step that no transition on the path completed as `failed`, in the transaction of the entering transition; a path into the abort terminal therefore needs no `completes` clause for such a step, and a clause it does carry is that step's single completion. No completion on a path, or two, fails validation.
** *Class admits the completion*: `as done` is admitted only on an outcome of class `SUCCESS` or a decision outcome of class `APPROVED` (an operator-authorized completion, for example the `automatic-review` force-done); `as skipped` only on `SUCCESS` or `SKIP`; `as skipped_by_operator` only on a decision outcome of class `APPROVED` (the operator settles a step whose evidence is missing or failed, for example the wait-region steps under an approved `review_barrier_gap`, `ci_failure_override`, or `wait_cancelled_override`), never on a primitive outcome. A `FAILURE`, `INDETERMINATE`, or `ABORTED` outcome never completes a step `done`, and a composite primitive whose sub-operations report any `FAILURE` outcome yields a `FAILURE` outcome itself.
** *Denied and aborted paths*: Every outcome of every registered `DecisionSpec` declares exactly one `DecisionOutcomeClass` (xref:02-units-jobs-decisions.adoc#decision-registry[Decision Registry & Typed Escalation Outcomes]). No transition on a path that follows a `DENIED` or `ABORTED` outcome completes a step `done` or `skipped_by_operator`, and such an outcome never enters a state running the decision's approved-action primitive; an `ABORTED` outcome must target the unit's `aborted` terminal state.
10. *Approving Decision Outcomes Execute the Approved Action*: Every `escalate` outcome of class `APPROVED` must target an `auto` state whose primitive is the one the `DecisionSpec` names for that outcome in `approvedActions`, so the approved action is performed in the transaction that applies the resolution. It must never target a terminal state, a sub-workflow terminal (`done`/`completed`) state, or a state that merely continues the flow without executing the action (a step selector, a dispatch or wait state, or a `link` state). An `APPROVED` outcome without an `approvedActions` entry fails validation. Example: `loop_limit_extension` `extended -> finalize.allocate-after-extension` (`auto plan.allocate-remediation-tasks`), never `extended -> execute.dispatch-ready`.
11. *No Self-Loop or Cycle Without an Awaited Event*: Every primitive declares the external event it waits for (`Primitive.awaitedEvent()`, xref:05-primitives.adoc#primitive-spi-contract[`Primitive<P, R>` SPI Contract]). A self-loop (an outcome transition that targets its own state) or a cycle of transitions fails validation when every primitive executed on it (`do`, `auto`, `job`, and `composite_job` primitives, including those of a sub-workflow called on the cycle) declares `NONE`: such a loop re-runs without any event that could change its outcome. A cycle passes when at least one of its primitives awaits `JOB_SETTLED`, `OPERATOR_DECISION`, `MODEL_SUBMIT`, `HEAD_CHANGE`, or `TIME`.
12. *Link Relation & Form Compatibility*: For every `LinkSpec`, the form (the declared `[form]`, or `LinkRelation.defaultForm()` when it is omitted) must be one of `rel.allowedForms()` (<<_ast_node_model>>, link:../hypermedia-format/02-link-grammar.adoc#_closed_link_relations_rel[Hypermedia Format Specification, Closed Link Relations]). The form must also fit the link: `submit` declares a non-empty parameter list; `noop`, `wait`, and `dispatch` declare none; `escalate` declares the action `escalate <decision>`, and that action is admitted only with the form `escalate`; its free parameters must equal the parameter schema its decision kind declares in the Decision Registry (possibly empty); the action `dispatch <entity_kind> role <role_key>` is admitted only with the form `dispatch`. A violation fails validation with the source file and line.

[#project-owned-units]
=== Project-Owned Workflow Units & Compiled Invariants
_See Requirement link:../../Requirements.adoc#PM-IMPL-5[PM-IMPL-5: Declarative Workflow DSL & Automated Diagram Generation]_

_See Requirement link:../../Requirements.adoc#PM-EXT-7[PM-EXT-7: Configuration Versioning, Validation & Zero-CoT Upgrades]_

The process a project runs is visible to and owned by its team: the units that define it are files in the repository, reviewed like code. The guarantees of the process stay server-enforced, because every unit is verified on every load against rules and invariants compiled into the binary, and a unit can only compose compiled primitives and enrolled scripts.

1. *Which Units Are Project-Owned*: Every phase file of the plan workflow (`init.pmwf`, `refine.pmwf`, `outline.pmwf`, `plan.pmwf`, `execute.pmwf`, `land.pmwf`, `finalize.pmwf`) and every sub-workflow unit (`scope subworkflow`, for example `git-commit-and-push.pmwf`) lives in the project as `.marshall/config/<file>.pmwf` (git-tracked, owner class *team*, link:../store-schemas/07-project-configuration.adoc#project-configuration-store[Store Schemas Specification, Project Configuration]). The binary ships the *default* units (`classpath:/workflows/`); `setup.prepare-workspace` copies them on the project's first run, each copy carrying `derived_from <n>` (the default revision it was copied from) in its header. The workspace unit `project-setup.pmwf` and the epic unit `epic.pmwf` are compiled only and never project-owned; they call the binary's default sub-workflows, never the project's copies, so a failing project unit never blocks the setup workflow that repairs it or the epic ledger.
2. *Load & Verification*: Every computation of the effective configuration (`pm_start`, every phase handshake, `pm-mcp enter`, `setup.reconcile-config`; cached by the units' digests) parses the project's units, merges the plan workflow unit (xref:02-units-jobs-decisions.adoc#workflow-definition-units[Workflow Definition Units (Multi-File Workflows)]), and verifies it with Rules 1–12 (<<static-verification-rules>>; Rule 5 only for default units), with `FinalizePipelineValidator` over the project's finalize roster (link:../phase-workflows/05-finalize-review-metrics-descriptor.adoc#_build_time_startup_static_invariant_verification_finalizepipelinevalidator[Phase Workflows Specification]), and with the compiled invariants (item 5). A unit at another DSL `format` is refused with `format_outdated` or `format_newer` naming the file. A missing, unparsable, or violating unit is never replaced by the default unit (no silent fallback): it is a verification finding in the workspace scope naming the file, the line, and the violated rule or invariant, and it refuses the project's plan lifecycle (`pm_start` and every phase handshake refuse with `config_invalid` naming the file and line) until the unit verifies. The verification outcome of every unit is published in `pm://project/config` (`units[]`).
3. *Project Steps Bind to Enrolled Scripts*: A unit never introduces executable code: every primitive it names must be registered in the binary (Rule 1). A project finalize step is a state running the compiled primitive `project.run-step` for its step id; it exists only when three owners agree: the unit contains the step, `workflow.json` `plan.phase-6-finalize.project_steps` declares its descriptor (`order`, `lane`, `project_script_id`), and the operator's enrolment binds the `project_script` with `worker_kind: finalize_step` and `target` equal to the step id (link:../domain-extensions/05-recipes-and-extension-points.adoc#_out_of_process_project_script_jobs_project_script_domain_workers[Domain Extensions Specification, `project_script`]). A disagreement is the verification finding `project_step_unbound`.
4. *Bound Unit Copy per Plan*: At `pm_start` the engine binds the verified unit sources to the plan (a copy in the plan's metadata directory, digest-recorded, part of `config_hash`); a plan always executes its bound copy. A changed project unit raises `config_drift` at the next phase handshake (link:../domain-extensions/06-configuration.adoc#config-snapshot[Domain Extensions Specification, Three-Surface Synchronization]); `adopt` replaces the bound copy only when the new unit verifies and still contains the plan's current state and every state of its `call_stack`, otherwise only `keep` is offered.
5. *Compiled Invariants*: The following invariants are compiled into the binary; no configuration key and no unit edit can switch one off, and every one of them is checked over the merged project unit on every load (`INV-*` identifiers name the invariant in findings):
+
[cols="1,2,5"]
|===
|Invariant |Unit |Check

|`INV-GEN-1`
|all
|Rules 1–12 hold; the unit's `scope`, start state, and terminal states equal those of the default unit (`plan.aborted` the only abort terminal, reachable from every phase; `completed` reached only through `archive-plan`).

|`INV-GEN-2`
|all
|Decision registry: every `escalate` names a kind registered in the binary's Decision Registry (xref:02-units-jobs-decisions.adoc#decision-registry[Decision Registry & Typed Escalation Outcomes]) with its compiled outcome set; every state that performs an approved action of a registered decision is reachable only through that decision's `APPROVED` outcome, never through a routine link or an `auto` transition; every operator-authorizing gate of the default unit (outline approval, deviation `accept`, `force-gate`, merge-authorization gap classes, `loop_limit_extension`, `abort-plan`, `land-verification-failed`, `archive-with-open-pr`) is present with its decision kind.

|`INV-GEN-3`
|all
|Condition overlays unchanged (<<condition-overlays>>): a unit declares no condition and targets none (Rule 3); the engine raises `blocked`, `footprint_violation`, `dispatch_incomplete`, `config_drift`, `store_tampered`, and `guard_indeterminate` for project units exactly as for default units, and no unit state suppresses them.

|`INV-GEN-4`
|all
|No guarantee weakening: no path of the unit realizes a removed option: a plan without a worktree, a direct commit to the base branch, a local-only run while a push remote exists, a forced `light` lane, a disabled lane ratchet or premise revalidation, a Q-Gate `off` or `once` mode, a lowered confidence threshold, an admin merge past branch protection, a server-minted merge grant, an auto-rebase over overlapping changes, a skipped self-review by scope, or a missing mandatory finalize step (the specific checks are the rows below).

|`INV-INIT-1`
|`init`
|`init.classify` runs the recipe scorer and the planning-lane router on every path from the start state (except the `recipe`, `local_changes`, and `pr` sources; the `pr` source is fixed to the `adopt` lane by `pm_start`); the router applies the team's `minimum_lane`; `main_sha` is stamped before `init.ready`.

|`INV-INIT-2`
|`init`, `outline`
|Every path of a `light` plan into `outline.light` passes `init.light-gate`, and every re-dispatch of `outline.light` passes `outline.light-regate`; the `light -> deep` ratchet transitions of the default unit are present.

|`INV-REFINE-1`
|`refine`
|Every deep path passes `refine.baseline-reconcile` and `refine.analyze` with the compiled confidence aggregation and the single re-dispatch ceiling; a lesson-derived deep plan passes `refine.qgate`.

|`INV-OUTLINE-1`
|`outline`
|`outline.review` lies on every path into `phase-4-plan`, and its only transition into `phase-4-plan` is the operator-authorizing outline approval; no link, continuation policy, or `auto` transition leaves it towards Phase 4.

|`INV-OUTLINE-2`
|`outline`
|Q-Gate presence: every deep-lane path from an accepted outline submission to `outline.review` passes `outline.qgate`, except through the compiled surgical-bypass predicate; `OutlineValidator` runs on every outline submission.

|`INV-PLAN-1`
|`plan`
|Every path into Phase 5 passes task derivation with the DAG check, `plan.qgate`, and the manifest composition (`ExecutionManifestResolver`).

|`INV-EXEC-1`
|`execute`, `land`
|Worktree: `execute.worktree-init` (or `land.worktree-init`, or for an `adopt` plan `finalize.adopt-pr`, which creates the worktree on the pull request's head branch) precedes every task dispatch, commit, and build of the plan; `in_worktree` guards every `loop_back` link and nothing else guards it.

|`INV-EXEC-2`
|`execute`, `land`
|Footprint checks: every task `submit` runs the task footprint check, and every commit goes through `call git.commit_and_push`, whose Stage 4 footprint check is present (`INV-GIT-1`); after every deliverable the chain-tail commit and the per-deliverable module build run; every path into Phase 6 passes the verify sweep and `execute.exit-barrier` (or `land.exit-barrier`), except the `adopt` lane, whose only path into Phase 6 is `init.ready` -> `finalize.adopt-pr` and whose verification is the finalize pipeline (`pre-push-quality-gate` and the push freshness gate, link:../phase-workflows/02-planning-lanes-phases-3-4.adoc#adopt-lane[Phase Workflows Specification, Adopt Lane]).

|`INV-FIN-1`
|`finalize`
|Mandatory steps: every step of lane class `core` or `derived-state` of the default roster (`sync-baseline`, `pre-push-quality-gate`, `push`, `create-pr`, `ci-verify`, `branch-cleanup`, `preference-emitter`, `record-metrics`, `print-phase-breakdown`, `emit-landing`, `archive-plan`, `architecture-refresh`) is present, with its catalogue `order` and descriptor.

|`INV-FIN-2`
|`finalize`
|Ordering bands: `FinalizePipelineValidator` rules 1–8 hold over the project roster (push barrier at `11` directly after `pre-push-quality-gate` at `10`, source mutation only below `70`, post-run review band, read-before-destroy, no refusing guard after the merge).

|`INV-FIN-3`
|`finalize`
|Completion exactly once: every begun step is completed exactly once on every path, never `done` on a failing, denied, or aborted path (Rule 9).

|`INV-FIN-4`
|`finalize`
|Push barrier: the only push of the branch before the pull request is `finalize.push-barrier`, and every push runs through `git.commit_and_push` with its freshness and branch-parity stages; a plan with a push remote reaches `finalize.create-pr`, except an `adopt` plan, whose pull request already exists and whose `create-pr` step is completed `skipped`.

|`INV-FIN-5`
|`finalize`
|Pre-merge barrier: every path to the merge passes the finalize spend check and `finalize.merge-gate` with both barrier predicates and the gap-class decisions; no path merges without them, and none mints a grant.

|`INV-FIN-6`
|`finalize`
|Branch cleanup and landing: `branch-cleanup` (`70`) removes the worktree in the transaction that clears `worktree_path`; `emit-landing` (`1000`) runs for every plan and is never pruned for an epic-emitted plan; `archive-plan` (`1100`) is the last step and the only path to `completed`.

|`INV-GIT-1`
|`git.commit_and_push`
|The six stages run in order on every path: worktree inspection, `.gitignore`-aware artifact detection, message resolution with Conventional Commit validation and the operator's trailer, stage-and-commit with the footprint check, the push-policy evaluation with the freshness gate and branch parity, and the terminal outcome set of the default unit.
|===
+
The invariants are defined against the default units: every invariant holds for them, and `DefaultUnitInvariantTest` verifies that at build time together with a fixture set of weakened units that each invariant must refuse.
6. *Offered Default Changes*: A new binary never changes a project unit. Its default-unit changes are offered in `setup.offer-new-steps` as a three-way diff from the unit's `derived_from` revision to the current default revision (link:../domain-extensions/06-configuration.adoc#config-upgrades[Domain Extensions Specification, Upgrade Path]); a unit whose `derived_from` is older than the default revision is valid and runs unchanged as long as it verifies.
