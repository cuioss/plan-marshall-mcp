== Reference Workflow DSL Excerpt (`finalize.pmwf`)
_Implementation watch: link:../../implementation-watch/phase-workflows.adoc#PM-WATCH-PHASE-31[PM-WATCH-PHASE-31]_

Every state the excerpt defines is catalogued in the link:../state-catalogue/04-plan-finalize-terminal.adoc#state-catalogue-finalize[State Catalogue, Phase 6: `finalize`].

[source,text]
----
workflow finalize version 1 format 1
scope plan
start finalize.settle-commit

state finalize.settle-commit
  phase finalize
  call git.commit_and_push (message_mode=step_or_fallback, push_mode=post_pr_conditional)
    committed_local      -> finalize.next-step
    committed_and_pushed -> finalize.review-cap-check
    pushed_clean         -> finalize.review-cap-check
    in_sync              -> finalize.review-cap-check
    clean                -> finalize.next-step
    aborted              -> finalize.commit-aborted

state finalize.push-barrier
  phase finalize
  call git.commit_and_push (message_mode=none_assert_clean, push_mode=if_push_remote)
    committed_local      -> finalize.create-pr
      completes default:push as skipped
    committed_and_pushed -> finalize.create-pr
      completes default:push as done
    pushed_clean         -> finalize.create-pr
      completes default:push as done
    in_sync              -> finalize.create-pr
      completes default:push as skipped
    clean                -> finalize.create-pr
      completes default:push as skipped
    aborted              -> finalize.commit-aborted

state finalize.commit-aborted
  phase finalize
  link retry retry [noop] ()
    do plan.resume-aborted-call
      settle_commit     -> finalize.settle-commit
      push_barrier      -> finalize.push-barrier
  link escalate abort [escalate] (reason: text[1..500])
    escalate abort-plan
      aborted           -> plan.aborted
      resumed           -> finalize.commit-aborted

state finalize.review-cap-check
  phase finalize
  auto review.check-structural-caps
    within_caps         -> finalize.concurrent-wait
    cap_exceeded        -> finalize.review-structural

state finalize.concurrent-wait
  phase finalize
  composite_job finalize.watch-ci-and-bots
    all_green                -> finalize.merge-gate
      completes default:ci-verify as done
      completes plan-marshall:automatic-review as done
      completes default:sonar-roundtrip as done
    findings_open            -> finalize.triage
    ci_failed                -> finalize.ci-failed
    bot_refused_awaitable    -> finalize.review-window-wait
    bot_refused_hard         -> finalize.review-escalate
    bot_refused_structural   -> finalize.review-structural
    bot_no_unreviewed_commit -> finalize.review-full-trigger
    bot_unproven             -> finalize.review-escalate
    roster_unestablished     -> finalize.review-escalate
    cancelled                -> finalize.wait-cancelled

state finalize.triage
  phase finalize
  link submit triage:<finding_id> [submit] (disposition: enum[FIX|SUPPRESS|ACCEPT], reply_text: text[1..4000], requirement_refs: list<text[1..120]>?, population: text[1..2000]?)
    do findings.triage
      open_remaining      -> finalize.triage
      triaged_with_fix    -> finalize.settle-commit
      triaged_without_fix -> finalize.next-step
        completes default:ci-verify as done
        completes plan-marshall:automatic-review as done
        completes default:sonar-roundtrip as done
  link loop_back loop_back [submit] (remediation_summary: text[1..2000])
    guard in_worktree
    do plan.allocate-remediation-tasks
      allocated         -> execute.dispatch-ready
      footprint_review  -> execute.deviation-review
      limit_reached     -> finalize.loop-limit-reached

state finalize.review-structural
  phase finalize
  link submit triage:<finding_id> [submit] (disposition: enum[FIX|SUPPRESS|ACCEPT], reply_text: text[1..4000], requirement_refs: list<text[1..120]>?, population: text[1..2000]?)
    do findings.triage
      open_remaining      -> finalize.review-structural
      triaged_with_fix    -> finalize.settle-commit
      triaged_without_fix -> finalize.review-cap-check

state finalize.review-window-wait
  phase finalize
  facts review.window-status
  job review.await-rate-window
    succeeded           -> finalize.review-retrigger
    failed              -> finalize.review-escalate
    refused             -> finalize.review-escalate
    cancelled           -> finalize.wait-cancelled
    timed_out           -> finalize.review-escalate
    killed              -> finalize.review-escalate
    indeterminate       -> finalize.review-escalate

state finalize.review-retrigger
  phase finalize
  auto review.retrigger-unproven
    triggered           -> finalize.review-cap-check
    window_closed       -> finalize.review-window-wait

state finalize.review-escalate
  phase finalize
  facts review.window-status
  link retry retry:<bot_kind> [noop] ()
    guard predicate:review.window-open
    do review.retrigger-bot
      triggered         -> finalize.review-cap-check
      window_closed     -> finalize.review-window-wait
  link escalate escalate [escalate] (reason: text[1..500])
    escalate review_barrier_gap
      approved          -> finalize.merge-gate
        completes default:ci-verify as skipped_by_operator
        completes plan-marshall:automatic-review as skipped_by_operator
        completes default:sonar-roundtrip as skipped_by_operator
      denied            -> plan.aborted
        completes default:ci-verify as failed
        completes plan-marshall:automatic-review as failed
        completes default:sonar-roundtrip as failed

state finalize.review-full-trigger
  phase finalize
  auto review.post-full-trigger
    triggered           -> finalize.review-cap-check
    already_triggered   -> finalize.review-escalate
    window_closed       -> finalize.review-window-wait

state finalize.spend-escalate
  phase finalize
  facts finalize.spend-ratio
  link escalate escalate [escalate] (reason: text[1..500])
    escalate finalize_spend_overrun
      approved          -> finalize.merge-gate
      denied            -> plan.aborted
        completes default:branch-cleanup as failed

state finalize.emit-landing-failed
  phase finalize
  facts finalize.landing-status
  link retry retry [noop] ()
    do finalize.retry-emit-landing
      verified          -> finalize.next-step
        completes default:emit-landing as done
      failed            -> finalize.emit-landing-failed
  link escalate escalate [escalate] (reason: text[1..500])
    escalate landing_delivery_failure
      approved          -> finalize.retry-landing-delivery
      denied            -> finalize.emit-landing-failed

state finalize.retry-landing-delivery
  phase finalize
  auto finalize.retry-landing-delivery
    delivered           -> finalize.next-step
      completes default:emit-landing as done
    pending             -> finalize.next-step
      completes default:emit-landing as done

state finalize.ci-failed
  phase finalize
  skill pm/finalize/ci-triage
  facts ci.failure-summary
  link fix fix_in_branch [submit] (fix_summary: text[1..2000])
    do finalize.record-in-branch-fix
      recorded          -> finalize.settle-commit
      no_changes        -> finalize.ci-failed
  link loop_back loop_back [submit] (remediation_summary: text[1..2000])
    guard in_worktree
    do plan.allocate-remediation-tasks
      allocated         -> execute.dispatch-ready
      footprint_review  -> execute.deviation-review
      limit_reached     -> finalize.loop-limit-reached
  link escalate escalate [escalate] (reason: text[1..500])
    escalate ci_failure_override
      approved          -> finalize.merge-gate
        completes default:ci-verify as skipped_by_operator
        completes plan-marshall:automatic-review as skipped_by_operator
        completes default:sonar-roundtrip as skipped_by_operator
      denied            -> plan.aborted
        completes default:ci-verify as failed
        completes plan-marshall:automatic-review as failed
        completes default:sonar-roundtrip as failed

state finalize.wait-cancelled
  phase finalize
  facts finalize.wait-status
  link retry retry [noop] ()
    do finalize.restart-watch
      started           -> finalize.concurrent-wait
  link escalate escalate [escalate] (reason: text[1..500])
    escalate wait_cancelled_override
      approved          -> finalize.merge-gate
        completes default:ci-verify as skipped_by_operator
        completes plan-marshall:automatic-review as skipped_by_operator
        completes default:sonar-roundtrip as skipped_by_operator
      denied            -> plan.aborted
        completes default:ci-verify as failed
        completes plan-marshall:automatic-review as failed
        completes default:sonar-roundtrip as failed

state finalize.loop-limit-reached
  phase finalize
  facts plan.loop-status
  link escalate escalate [escalate] (reason: text[1..500])
    escalate loop_limit_extension
      extended          -> finalize.allocate-after-extension
      aborted           -> plan.aborted
        completes default:ci-verify as failed
        completes plan-marshall:automatic-review as failed
        completes default:sonar-roundtrip as failed

state finalize.allocate-after-extension
  phase finalize
  auto plan.allocate-remediation-tasks
    allocated         -> execute.dispatch-ready
    footprint_review  -> execute.deviation-review
    limit_reached     -> finalize.loop-limit-reached

state plan.aborted terminal aborted failure
  phase aborted
----

* *Hard CI failure (`PM-WF-9`)*: `finalize.ci-failed` offers both remediation paths. `fix_in_branch` (rel `fix`) is followed after the model has applied the fix in the worktree; `finalize.record-in-branch-fix` records `fix_summary` and routes through `finalize.settle-commit` (`call git.commit_and_push` with `push_mode=post_pr_conditional`), whose `committed_and_pushed` outcome re-enters `finalize.concurrent-wait` (through `finalize.review-cap-check`) so CI and review bots re-evaluate the new `HEAD`. A clean worktree yields `no_changes` and leaves the plan in `finalize.ci-failed`. `loop_back` returns to `execute` for cross-task remediation (xref:07-loop-back-diagrams-verification.adoc#loop-back-bounds[Worktree-Constrained Loop-Back & Iteration Bounds]).
* *Pre-wait cap gate*: Before `finalize.concurrent-wait` is entered, `finalize.review-cap-check` measures the diff (changed files, diff characters) against the structural caps each required review bot declares in the review-bot registry (link:../phase-workflows.adoc[Phase Workflows Specification]). An exceeded cap records the bot as `refused` with mode `size` pre-emptively, names the cap and the measurement in `facts`, and routes to `finalize.review-structural` instead of waiting; under every cap the flow is unchanged.
* *Review-bot refusal modes*: The composite outcomes of `finalize.watch-ci-and-bots` separate the refusal modes, because a single rate-limit outcome cannot express them. `bot_refused_awaitable` (a rate window that reopens) enters `finalize.review-window-wait`, a job that waits for the registered window (the registered interval times the attempt ceiling; a published ETA is a floor that is re-checked, never skipped) and then re-triggers the unproven bots automatically. `bot_refused_hard` (weekly or hard quota) and `bot_unproven` (no review evidence for the current `HEAD` after the wait) enter `finalize.review-escalate`. `bot_refused_structural` (size) enters `finalize.review-structural`, which never waits: it shows each structural refusal as a finding and offers `triage:<finding_id>` per finding, the link and primitive of `finalize.triage`, where `FIX` splits (when a path partition, for example test sources versus the rest, brings each part under the cap, the finding proposes that concrete split), `ACCEPT` accepts the refusal, and `SUPPRESS` disables the bot for this pull request. Once every structural finding is triaged, a `FIX` takes the fix path through `finalize.settle-commit`; otherwise the flow returns through `finalize.review-cap-check`, which no longer counts the accepted or disabled bots, into the wait region, where any other open finding is triaged in `finalize.triage` (`findings_open`). `bot_no_unreviewed_commit` enters `finalize.review-full-trigger`, which posts the registry's full-review trigger verb.
* *Full-review trigger*: `finalize.review-full-trigger` posts the registry's `full` trigger verb for the bots that reported no unreviewed commit, once per bot and `HEAD`; a repeated notice after that trigger yields `already_triggered` and escalates, and a closed rate window routes to `finalize.review-window-wait` (link:../phase-workflows/05-finalize-review-metrics-descriptor.adoc#automatic-review[Phase Workflows Specification], Refusals & Rate Windows).
* *Unestablished roster*: `roster_unestablished` (the required set is empty because the roster question was never put) enters `finalize.review-escalate`; with no bot to re-trigger, its only way on is the `review_barrier_gap` escalation.
* *Finalize spend overrun*: Before the pre-merge barrier, `finalize.merge-gate` compares the cumulative finalize spend with the error column of the plan's budget anchor (link:../dispatch-and-effort/02-dispatch-observability-and-spend.adoc#spend-accounting-budget-anchors[Dispatch & Effort Specification]); an overrun enters `finalize.spend-escalate`, whose approved escalation records the acknowledgement and returns to `finalize.merge-gate`.
* *Merge path*: The merge-path states after `finalize.merge-gate` are not part of the excerpt; they are catalogued in the link:../state-catalogue/04-plan-finalize-terminal.adoc#state-catalogue-finalize[State Catalogue]. Their link states hold no merge lease (it is released before they are entered), offer a way back into the merge path (`retry`, `resume_wait`, `integrate`, `continue_rebase`), the request `abort` (`abort-plan`, except in `finalize.merge-conflict`, which returns through `abort_rebase`), and, where the plan cannot merge (`finalize.merge-refused`, `finalize.merge-queue-exit`, `finalize.merge-gap`), the link `wind_down` (rel `escalate`, `wind-down-unmerged`), whose approval enters the `auto` state `finalize.wind-down` (Rule 10) and whose denial returns to the raising state. A denied gap escalation of `finalize.merge-gap` returns to `finalize.merge-gap` and never aborts the plan.
* *Unified triage exit*: `finalize.triage` offers `triage:<finding_id>` for every open finding and `loop_back`. The triage primitive `findings.triage` decides the exit automatically once the last open finding is triaged: `open_remaining` stays in the state; `triaged_with_fix` (at least one `FIX`) takes the in-branch fix path through `finalize.settle-commit` (a `FIX` that needs cross-task remediation is taken through `loop_back` instead); `triaged_without_fix` enters `finalize.next-step` and completes the three wait-region steps as `done`.
* *Landing delivery failure*: An I/O failure of the `emit-landing` append enters `finalize.emit-landing-failed`. `retry` re-appends and re-verifies the envelope; the approved escalation enters the `auto` state `finalize.retry-landing-delivery`, which repeats the landing delivery once more and continues the pipeline, with `landing_delivery: pending` when that delivery fails again, keeping the envelope in the plan directory for the epic's shipped-without-landing reconciliation. Neither link refuses or strands the merged plan (link:../phase-workflows.adoc[Phase Workflows Specification], validator rule 7); a denied escalation leaves the plan in the state, never aborts it.
* *Window-gated re-trigger*: `finalize.review-escalate` offers `retry:<bot_kind>` only while the predicate `review.window-open` holds, so a trigger is never posted inside a closed rate window (a trigger in a closed window extends it; closing and reopening the PR does not reset it). The `<bot_kind>` placeholder expands over the unproven bots of `required_bots ∪ optional_bots` (only the required ones gate; link:../phase-workflows/05-finalize-review-metrics-descriptor.adoc#automatic-review[Phase Workflows Specification, Refusals & Rate Windows] item 9) that have no review evidence for the current `HEAD`, selected after the round's own comment fetch and never by comment recency; a required bot that never published is selectable. A closed window at execution time routes to `finalize.review-window-wait`.
* *Implicit job links*: `finalize.concurrent-wait` is a `composite_job` state and therefore offers `wait:finalize.concurrent-wait` and `cancel:finalize.concurrent-wait` without declaring them (xref:02-units-jobs-decisions.adoc#implicit-job-links[Implicit Links of Job States]). Following `cancel` yields the composite outcome `cancelled`, handled by `finalize.wait-cancelled`.
* *Loop-limit extension*: The approving outcome `extended` of `loop_limit_extension` enters `finalize.allocate-after-extension`, an `auto` state that runs `plan.allocate-remediation-tasks` for the `loop_back` refused at the limit, and never `execute.dispatch-ready` directly, which would dispatch without the fix tasks and envelope the extension granted (Rule 10, xref:07-loop-back-diagrams-verification.adoc#loop-back-bounds[Worktree-Constrained Loop-Back & Iteration Bounds]). Its outcomes take the same post-allocation path as every `loop_back` link.
* *Escalation outcomes*: Every `escalate` decision kind of the excerpt is registered with its closed outcome set and outcome classes (xref:02-units-jobs-decisions.adoc#decision-registry[Decision Registry & Typed Escalation Outcomes]).
* *Aborted commits*: An `aborted` outcome of `git.commit_and_push` in `finalize.settle-commit` or `finalize.push-barrier` enters `finalize.commit-aborted`, never the plan's abort terminal: `retry` re-enters the calling state (`plan.resume-aborted-call` returns the outcome naming it), and the link `abort` requests `abort-plan`, whose `aborted` outcome enters `plan.aborted` and whose `resumed` outcome returns to the state.
* *Step completion*: The `completes` clauses name the finalize steps each transition settles: `default:push` in `finalize.push-barrier` (`skipped` when the tree was clean, the push was not required, or the remote branch already held `HEAD`, `in_sync`), `default:ci-verify`, `plan-marshall:automatic-review`, and `default:sonar-roundtrip` (the `22..40` wait region entered through `finalize.review-cap-check`) as `done` on `all_green` and on `triaged_without_fix`, as `failed` on every denied or aborted path to `plan.aborted`, and as `skipped_by_operator` on the `approved` outcome of `review_barrier_gap`, `ci_failure_override`, and `wait_cancelled_override` (each step record references the decision; Phase-Exit Completeness accepts it, and the barrier predicates and the `red_ci_gate` check still guard the merge), `default:branch-cleanup` on the denied spend escalation and in `finalize.wind-down` (`failed`), and `default:emit-landing` in `finalize.emit-landing-failed` and `finalize.retry-landing-delivery`. A step begun but not completed when `plan.aborted` is entered through `abort-plan` (for example `default:push` after an aborted commit) is recorded `failed` on entry (Rule 9). A `loop_back` outcome is `UNSETTLED` and completes nothing (xref:05-primitives.adoc#primitive-outcome-classes[Outcome Classes & Step Records]). What each approval records at `finalize.merge-gate` (a grant of which gap class, or an acknowledgement only) is the grant mapping of the Decision Registry (xref:02-units-jobs-decisions.adoc#decision-grant-mapping[Decision grant mapping]).
* *Cross-file targets*: `execute.dispatch-ready` is declared in `execute.pmwf`; the target is valid because all phase files form one definition unit (xref:02-units-jobs-decisions.adoc#workflow-definition-units[Workflow Definition Units (Multi-File Workflows)]).
